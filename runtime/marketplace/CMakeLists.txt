cmake_minimum_required(VERSION 3.20)
project(CrimsonMarketplace LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Marketplace modules require Windows x64 and MSVC")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(MINHOOK_ROOT "" CACHE PATH "Pinned MinHook 1.3.4 checkout")
if(NOT EXISTS "${MINHOOK_ROOT}/CMakeLists.txt")
  message(FATAL_ERROR "Provide MINHOOK_ROOT from dependencies.lock.json")
endif()
add_subdirectory("${MINHOOK_ROOT}" minhook)
foreach(module IN ITEMS Blackstar SleepCooldown SleepDuration)
  if(module STREQUAL "Blackstar")
    set(source blackstar/blackstar_mod.cpp)
  elseif(module STREQUAL "SleepCooldown")
    set(source sleep/sleep_mod.cpp)
  else()
    set(source sleep/duration_mod.cpp)
  endif()
  add_library(Crimson${module} SHARED ${source})
  target_link_libraries(Crimson${module} PRIVATE minhook bcrypt)
  target_compile_definitions(Crimson${module} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  set_target_properties(Crimson${module} PROPERTIES SUFFIX ".asi")
endforeach()
enable_testing()
add_executable(marketplace-policy-tests tests/policy_tests.cpp)
target_compile_options(marketplace-policy-tests PRIVATE /UNDEBUG)
add_test(NAME marketplace-policy COMMAND marketplace-policy-tests)
