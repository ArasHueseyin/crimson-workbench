  elseif(module STREQUAL "GlideSpeed")
    set(source glide/glide_mod.cpp)
  elseif(module STREQUAL "MountDismount")
    set(source dismount/mount_dismount_mod.cpp)
  elseif(module STREQUAL "AutoLoot")
    set(source auto-loot/auto_loot_mod.cpp)

    set(source world-controls/world_mod.cpp world-controls/control_pipe.cpp)

    set(source photo-mode/photo_mod.cpp photo-mode/photo_pipe.cpp ../repair/src/hash.cpp)
  elseif(module STREQUAL "Teleport")
    set(source teleport/teleport_mod.cpp teleport/teleport_pipe.cpp ../repair/src/hash.cpp)
  else()
    set(source sleep/duration_mod.cpp)
  endif()
  add_library(Crimson${module} SHARED ${source})
  target_link_libraries(Crimson${module} PRIVATE minhook bcrypt)
  if(module STREQUAL "WorldControls" OR module STREQUAL "PhotoMode" OR module STREQUAL "Teleport")
    target_link_libraries(Crimson${module} PRIVATE advapi32)
    target_compile_options(Crimson${module} PRIVATE /W4 /WX)
  endif()
  if(module STREQUAL "Teleport")
    target_link_libraries(Crimson${module} PRIVATE crypt32 shell32 ole32 uuid)
  endif()
  if(module STREQUAL "PlayerJump" OR module STREQUAL "PlayerSpeed" OR module STREQUAL "PreserveConsumables" OR module STREQUAL "PlayerStamina" OR module STREQUAL "ForcePalmHeight" OR module STREQUAL "AxiomReach" OR module STREQUAL "GlideSpeed")
    target_link_libraries(Crimson${module} PRIVATE shell32 ole32 uuid)
  endif()
  if(module STREQUAL "PlayerStamina" OR module STREQUAL "ForcePalmHeight" OR module STREQUAL "AxiomReach" OR module STREQUAL "GlideSpeed")
    target_compile_options(Crimson${module} PRIVATE /W4 /WX)
  endif()
  if(module STREQUAL "PreserveConsumables" OR module STREQUAL "MountProtection" OR module STREQUAL "FreeCrafting" OR module STREQUAL "ForcePalm" OR module STREQUAL "MountDismount" OR module STREQUAL "AutoLoot")
    target_compile_options(Crimson${module} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4 /WX>)
  endif()
  target_compile_definitions(Crimson${module} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  set_target_properties(Crimson${module} PROPERTIES SUFFIX ".asi")
endforeach()

# Original-game verification is a separate, explicit local research mode.
# Public CI uses only synthetic fixtures and needs no extracted game records.
set(CRIMSON_PRIVATE_CONSUMABLE_EVIDENCE_HEADER "" CACHE FILEPATH "Optional private consumable verification header")
set(CRIMSON_PRIVATE_GLIDE_FIXTURE_HEADER "" CACHE FILEPATH "Optional private glide verification header")
set(CRIMSON_PRIVATE_FORCE_PALM_HEIGHT_FIXTURE_HEADER "" CACHE FILEPATH "Optional private Force Palm height verification header")
foreach(fixture IN ITEMS CONSUMABLE_EVIDENCE GLIDE_FIXTURE FORCE_PALM_HEIGHT_FIXTURE)
  set(variable "CRIMSON_PRIVATE_${fixture}_HEADER")
  if(NOT "${${variable}}" STREQUAL "")
    if(NOT IS_ABSOLUTE "${${variable}}" OR NOT EXISTS "${${variable}}" OR IS_DIRECTORY "${${variable}}")
      message(FATAL_ERROR "${variable} must name an existing absolute private header")
    endif()
    if(fixture STREQUAL "CONSUMABLE_EVIDENCE")
      set(fixture_target marketplace-consumables-tests)
    elseif(fixture STREQUAL "GLIDE_FIXTURE")
      set(fixture_target marketplace-glide-tests)
    else()
      set(fixture_target marketplace-force-palm-height-tests)
    endif()
    target_compile_definitions(${fixture_target} PRIVATE "${variable}=\"${${variable}}\"")
  endif()
endforeach()
