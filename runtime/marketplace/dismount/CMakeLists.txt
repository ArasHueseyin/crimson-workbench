cmake_minimum_required(VERSION 3.20)
project(CrimsonMountDismountFixture LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Requires Windows x64 and MSVC")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(MINHOOK_ROOT "" CACHE PATH "Pinned MinHook 1.3.4 checkout")
add_subdirectory("${MINHOOK_ROOT}" minhook)
add_library(CrimsonMountDismount SHARED mount_dismount_mod.cpp)
target_link_libraries(CrimsonMountDismount PRIVATE minhook bcrypt)
target_compile_definitions(CrimsonMountDismount PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(CrimsonMountDismount PRIVATE /W4 /WX)
set_target_properties(CrimsonMountDismount PROPERTIES SUFFIX ".asi")
add_executable(marketplace-mount-dismount-tests ../tests/mount_dismount_tests.cpp)
target_compile_definitions(marketplace-mount-dismount-tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(marketplace-mount-dismount-tests PRIVATE /UNDEBUG /W4 /WX)
enable_testing()
add_test(NAME marketplace-mount-dismount COMMAND marketplace-mount-dismount-tests)
