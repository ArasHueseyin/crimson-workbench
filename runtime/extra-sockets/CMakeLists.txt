cmake_minimum_required(VERSION 3.24)
project(CrimsonExtraSockets LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64/MSVC required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(MINHOOK_ROOT "" CACHE PATH "Unmodified MinHook 1.3.4 source tree")
set(CRIMSON_LIVE_ITEMS_SHA256 "576dad0e64dfd352b89bca2dd2b031ff8256c824cdd279732cd1dd186ea74d05" CACHE STRING "SHA256 of the paired CrimsonLiveItems.asi")
string(LENGTH "${CRIMSON_LIVE_ITEMS_SHA256}" PEER_HASH_LENGTH)
if(NOT PEER_HASH_LENGTH EQUAL 64 OR NOT CRIMSON_LIVE_ITEMS_SHA256 MATCHES "^[0-9a-f]+$")
  message(FATAL_ERROR "CRIMSON_LIVE_ITEMS_SHA256 must be a lowercase SHA256 digest")
endif()
add_subdirectory("${MINHOOK_ROOT}" minhook)
add_library(CrimsonExtraSockets SHARED src/socket_mod.cpp)
target_link_libraries(CrimsonExtraSockets PRIVATE minhook bcrypt)
target_compile_definitions(CrimsonExtraSockets PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX CRIMSON_LIVE_ITEMS_SHA256="${CRIMSON_LIVE_ITEMS_SHA256}")
set_target_properties(CrimsonExtraSockets PROPERTIES SUFFIX ".asi")
