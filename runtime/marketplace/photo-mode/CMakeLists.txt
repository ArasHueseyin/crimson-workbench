cmake_minimum_required(VERSION 3.20)
project(CrimsonPhotoModeOwnedFixture LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Photo mode requires Windows x64 and MSVC")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(MINHOOK_ROOT "" CACHE PATH "Pinned MinHook 1.3.4 checkout")
if(NOT TARGET minhook)
  if(NOT EXISTS "${MINHOOK_ROOT}/CMakeLists.txt")
    message(FATAL_ERROR "Provide the pinned MINHOOK_ROOT")
  endif()
  add_subdirectory("${MINHOOK_ROOT}" minhook)
endif()
add_library(CrimsonPhotoMode SHARED photo_mod.cpp photo_pipe.cpp ../../repair/src/hash.cpp)
target_link_libraries(CrimsonPhotoMode PRIVATE minhook bcrypt advapi32)
target_compile_options(CrimsonPhotoMode PRIVATE /W4 /WX)
target_compile_definitions(CrimsonPhotoMode PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
set_target_properties(CrimsonPhotoMode PROPERTIES SUFFIX ".asi")
add_executable(marketplace-photo-mode-tests ../tests/photo_mode_tests.cpp)
target_compile_options(marketplace-photo-mode-tests PRIVATE /W4 /WX /UNDEBUG)
target_compile_definitions(marketplace-photo-mode-tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
enable_testing()
add_test(NAME marketplace-photo-mode COMMAND marketplace-photo-mode-tests)
