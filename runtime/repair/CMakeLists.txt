cmake_minimum_required(VERSION 3.24)
project(CrimsonRepairPrototype VERSION 0.23.0 LANGUAGES CXX)

if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This isolated execution harness requires Windows x64 and MSVC.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_library(crimson_repair_prototype SHARED src/plan.cpp src/hash.cpp)
target_include_directories(crimson_repair_prototype PUBLIC include PRIVATE src)
target_compile_definitions(crimson_repair_prototype PRIVATE CRIMSON_REPAIR_EXPORTS)
target_link_libraries(crimson_repair_prototype PRIVATE bcrypt)
set_target_properties(crimson_repair_prototype PROPERTIES OUTPUT_NAME CrimsonRepairPrototype)

add_library(crimson_repair_action STATIC src/action.cpp)
add_library(crimson_repair_item STATIC src/item.cpp)
target_link_libraries(crimson_repair_item PUBLIC crimson_repair_action)
target_include_directories(crimson_repair_item PUBLIC include)
add_executable(repair-item-tests tests/item_tests.cpp)
target_link_libraries(repair-item-tests PRIVATE crimson_repair_item)
target_include_directories(crimson_repair_action PUBLIC include)
add_library(crimson_repair_reader STATIC src/reader.cpp src/reader_win.cpp)
target_include_directories(crimson_repair_reader PUBLIC include)
target_link_libraries(crimson_repair_reader PUBLIC crimson_repair_action)
add_library(crimson_repair_writer STATIC src/writer.cpp src/writer_win.cpp)
target_include_directories(crimson_repair_writer PUBLIC include)
target_link_libraries(crimson_repair_writer PUBLIC crimson_repair_reader)
add_library(crimson_repair_transaction STATIC src/transaction.cpp)
target_include_directories(crimson_repair_transaction PUBLIC include)
target_link_libraries(crimson_repair_transaction PUBLIC crimson_repair_writer crimson_repair_item)
add_executable(repair-transaction-tests tests/transaction_tests.cpp)
target_link_libraries(repair-transaction-tests PRIVATE crimson_repair_transaction)
add_executable(repair-writer-tests tests/writer_tests.cpp)
target_link_libraries(repair-writer-tests PRIVATE crimson_repair_writer)
add_executable(repair-action-tests tests/action_tests.cpp)
target_link_libraries(repair-action-tests PRIVATE crimson_repair_action)
add_executable(repair-reader-tests tests/reader_tests.cpp)
target_link_libraries(repair-reader-tests PRIVATE crimson_repair_reader)
add_library(crimson_repair_lease STATIC src/lease.cpp)
add_library(crimson_repair_reference STATIC src/reference.cpp)
add_library(crimson_repair_registry STATIC src/registry.cpp)
target_include_directories(crimson_repair_registry PUBLIC include)
target_link_libraries(crimson_repair_registry PUBLIC crimson_repair_reference)
add_executable(repair-registry-tests tests/registry_tests.cpp)
target_link_libraries(repair-registry-tests PRIVATE crimson_repair_registry)
target_include_directories(crimson_repair_reference PUBLIC include)
target_link_libraries(crimson_repair_reader PUBLIC crimson_repair_reference)
target_include_directories(crimson_repair_lease PUBLIC include)
target_link_libraries(crimson_repair_reader PUBLIC crimson_repair_lease)
add_executable(repair-lease-tests tests/lease_tests.cpp)
target_link_libraries(repair-lease-tests PRIVATE crimson_repair_lease)

add_executable(repair-harness tests/harness.cpp src/hash.cpp src/pe.cpp)
add_executable(repair-registry-native tests/registry_native.cpp src/hash.cpp src/pe.cpp)
target_include_directories(repair-registry-native PRIVATE include src)
target_link_libraries(repair-registry-native PRIVATE crimson_repair_registry crimson_repair_transaction bcrypt)
target_include_directories(repair-harness PRIVATE include src)
target_link_libraries(repair-harness PRIVATE crimson_repair_prototype crimson_repair_reader crimson_repair_lease bcrypt)
foreach(t crimson_repair_prototype crimson_repair_action crimson_repair_reader crimson_repair_writer crimson_repair_transaction crimson_repair_lease crimson_repair_reference crimson_repair_registry crimson_repair_item repair-item-tests repair-writer-tests repair-transaction-tests repair-action-tests repair-reader-tests repair-lease-tests repair-registry-tests repair-registry-native repair-harness)
  target_compile_definitions(${t} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(${t} PRIVATE /W4 /WX /permissive- /utf-8 /EHsc)
endforeach()
enable_testing()
add_test(NAME repair-prototype-contract COMMAND repair-harness)
add_test(NAME repair-action COMMAND repair-action-tests)
add_test(NAME repair-reader COMMAND repair-reader-tests)
add_test(NAME repair-lease COMMAND repair-lease-tests)
add_test(NAME repair-registry COMMAND repair-registry-tests)
add_test(NAME repair-registry-native-manifest COMMAND repair-registry-native)
add_test(NAME repair-item COMMAND repair-item-tests)
add_test(NAME repair-writer COMMAND repair-writer-tests)
add_test(NAME repair-transaction COMMAND repair-transaction-tests)
