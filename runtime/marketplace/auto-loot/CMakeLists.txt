cmake_minimum_required(VERSION 3.20)
project(CrimsonAutoLootOwnedFixture LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Auto-loot requires Windows x64 and MSVC")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(MINHOOK_ROOT "" CACHE PATH "Pinned MinHook 1.3.4 checkout")
if(NOT TARGET minhook)
  if(NOT EXISTS "${MINHOOK_ROOT}/CMakeLists.txt")
    message(FATAL_ERROR "Provide the pinned MINHOOK_ROOT")
  endif()
  add_subdirectory("${MINHOOK_ROOT}" minhook)
endif()
add_library(CrimsonAutoLoot SHARED auto_loot_mod.cpp)
target_link_libraries(CrimsonAutoLoot PRIVATE minhook bcrypt)
target_compile_options(CrimsonAutoLoot PRIVATE /W4 /WX)
target_compile_definitions(CrimsonAutoLoot PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
set_target_properties(CrimsonAutoLoot PROPERTIES SUFFIX ".asi")
add_executable(marketplace-auto-loot-tests ../tests/auto_loot_tests.cpp)
target_compile_options(marketplace-auto-loot-tests PRIVATE /W4 /WX /UNDEBUG)
target_compile_definitions(marketplace-auto-loot-tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
enable_testing()
add_test(NAME marketplace-auto-loot COMMAND marketplace-auto-loot-tests)
