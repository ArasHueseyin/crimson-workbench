cmake_minimum_required(VERSION 3.24)
project(CrimsonExtraSockets LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64/MSVC required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(MINHOOK_ROOT "" CACHE PATH "Unmodified MinHook 1.3.4 source tree")
add_subdirectory("${MINHOOK_ROOT}" minhook)
add_library(CrimsonExtraSockets SHARED src/socket_mod.cpp)
target_link_libraries(CrimsonExtraSockets PRIVATE minhook bcrypt)
target_compile_definitions(CrimsonExtraSockets PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
set_target_properties(CrimsonExtraSockets PROPERTIES SUFFIX ".asi")
