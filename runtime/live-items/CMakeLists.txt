cmake_minimum_required(VERSION 3.24)
project(CrimsonLiveItemsProbe LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64/MSVC required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(MINHOOK_ROOT "" CACHE PATH "Unmodified MinHook 1.3.4 source tree")
add_subdirectory("${MINHOOK_ROOT}" minhook)
add_library(CrimsonLiveItemsProbe SHARED src/probe.cpp ../repair/src/hash.cpp)
target_include_directories(CrimsonLiveItemsProbe PRIVATE ../repair/src)
target_compile_definitions(CrimsonLiveItemsProbe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(CrimsonLiveItemsProbe PRIVATE minhook bcrypt)
set_target_properties(CrimsonLiveItemsProbe PROPERTIES SUFFIX ".asi")
add_library(CrimsonLiveItems SHARED src/runtime.cpp src/pipe.cpp ../repair/src/hash.cpp)
target_include_directories(CrimsonLiveItems PRIVATE ../repair/src)
target_compile_definitions(CrimsonLiveItems PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(CrimsonLiveItems PRIVATE minhook bcrypt advapi32)
set_target_properties(CrimsonLiveItems PROPERTIES SUFFIX ".asi")
add_executable(live-items-probe-tests tests/probe_tests.cpp)
target_compile_definitions(live-items-probe-tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
enable_testing()
add_test(NAME live-items-probe COMMAND live-items-probe-tests)
add_executable(live-items-grant-tests tests/grant_tests.cpp)
target_compile_definitions(live-items-grant-tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
add_test(NAME live-items-grants COMMAND live-items-grant-tests)
add_executable(live-items-native-fixture tests/native_fixture.cpp ../repair/src/hash.cpp)
target_include_directories(live-items-native-fixture PRIVATE ../repair/src)
target_compile_definitions(live-items-native-fixture PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(live-items-native-fixture PRIVATE bcrypt minhook)
add_executable(live-items-pipe-host tests/pipe_host.cpp src/pipe.cpp)
target_compile_definitions(live-items-pipe-host PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(live-items-pipe-host PRIVATE advapi32)
add_executable(live-items-loader-tests tests/loader_tests.cpp)
target_link_libraries(live-items-loader-tests PRIVATE winmm)
